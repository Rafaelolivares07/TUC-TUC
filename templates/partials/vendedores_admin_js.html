${d.error || 'Error'}

`; return; } if (!d.vendedores.length) { cont.innerHTML = '

Sin vendedores asignados aun.

'; return; } cont.innerHTML = d.vendedores.map(v => { const nombre = htmlVendedor(v.nombre || ''); const telefono = htmlVendedor(v.telefono || ''); const telAttr = attrVendedor(v.telefono || ''); const nombreAttr = attrVendedor(v.nombre || ''); return `

${nombre}

${telefono}

`; }).join(''); } catch(e) { cont.innerHTML = '

Error de conexion

'; } } async function agregarVendedor() { const tel = document.getElementById('vd-tel').value.trim().replace(/\D/g,''); const nombre = document.getElementById('vd-nombre').value.trim(); if (tel.length < 10) { toast('Telefono debe tener al menos 10 digitos', true); return; } try { const r = await fetch(window.VENDEDORES_API_BASE, { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ telefono: tel, nombre }) }); const d = await r.json(); if (d.ok) { toast('Vendedor agregado: ' + d.nombre); document.getElementById('vd-tel').value = ''; document.getElementById('vd-nombre').value = ''; document.getElementById('vd-autocomplete').classList.add('hidden'); cargarVendedores(); } else { toast(d.error || 'Error al agregar', true); } } catch(e) { toast('Error de conexion', true); } } async function eliminarVendedor(id) { if (!confirm('Quitar a este vendedor del negocio?')) return; try { const r = await fetch(window.VENDEDORES_API_BASE + '/' + id, { method: 'DELETE' }); const d = await r.json(); if (d.ok) { toast('Vendedor removido'); cargarVendedores(); } else toast(d.error || 'Error', true); } catch(e) { toast('Error de conexion', true); } } function urlDashboardVendedor(tel) { const limpio = (tel || '').replace(/\D/g,''); return `${window.location.origin}/vendedor${limpio ? '?tel=' + encodeURIComponent(limpio) : ''}`; } function telefonoWhatsAppVendedor(tel) { const limpio = (tel || '').replace(/\D/g,''); if (limpio.length === 10) return '57' + limpio; return limpio; } function abrirDashboardVendedor(tel) { window.open(urlDashboardVendedor(tel), '_blank'); } function compartirDashboardVendedor(tel, nombre) { const waTel = telefonoWhatsAppVendedor(tel); if (!waTel) { toast('El vendedor no tiene telefono para WhatsApp', true); return; } const url = urlDashboardVendedor(tel); const negocio = window.VENDEDORES_NEGOCIO_NOMBRE ? ` para ${window.VENDEDORES_NEGOCIO_NOMBRE}` : ''; const msg = `Hola ${nombre || ''}. Este es tu enlace al dashboard de vendedores TUC TUC${negocio}: ${url}`; window.open(`https://wa.me/${waTel}?text=${encodeURIComponent(msg)}`, '_blank'); } let _vdTimer = null; function onVdTelInput(val) { const tel = val.replace(/\D/g,''); buscarTercerosVendedor(tel, 4); } function onVdNombreInput(val) { buscarTercerosVendedor(val.trim(), 2); } function buscarTercerosVendedor(q, minimo) { clearTimeout(_vdTimer); if (q.length < minimo) { document.getElementById('vd-autocomplete').classList.add('hidden'); return; } _vdTimer = setTimeout(async () => { try { const r = await fetch('/api/vendedor/buscar-terceros?q=' + encodeURIComponent(q)); const d = await r.json(); const resultados = Array.isArray(d) ? d : (d.resultados || []); const box = document.getElementById('vd-autocomplete'); if (!resultados.length) { box.classList.add('hidden'); return; } box.innerHTML = resultados.map(t => `

` ).join(''); box.classList.remove('hidden'); } catch {} }, 300); } function selVd(tel, nombre) { document.getElementById('vd-tel').value = tel; document.getElementById('vd-nombre').value = nombre; document.getElementById('vd-autocomplete').classList.add('hidden'); } function htmlVendedor(value) { return String(value || '').replace(/[&<>"']/g, c => ({'&':'&','<':'<','>':'>','"':'"',"'":'''}[c])); } function attrVendedor(value) { return String(value || '').replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/\r?\n/g, ' '); }
